문제 8번이랑 마찬가지로 Recursion 이용하는 문제

```
#include <stdio.h>

int power(int base, int power_raised) {
    if(power_raised != 1) { // 지수에서 -1 해서 1될 때 까지 곱함
        return base * power(base, power_raised - 1);
    }
    else {
        return base;
    }

}

int main() {
    int base, pow;
    printf("밑수: ");
    scanf("%d", &base);

    printf("지수: ");
    scanf("%d", &pow);

    printf("%d^%d = %d", base, pow, power(base, pow));
}
```
